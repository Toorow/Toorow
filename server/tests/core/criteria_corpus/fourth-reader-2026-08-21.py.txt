import re


def _incomplete_if(path):
    """Les criteres d'acceptation d'une surface, VERBATIM."""
    text = _read(path)
    match = re.search(r"^##+\s*Incomplete if\s*$(.+?)(?=^##\s|\Z)", text, re.M | re.S)
    if not match:
        return []
    criteria = []
    for raw in re.split(r"\n(?=\s*(?:-\s|\d+\.\s))", match.group(1)):
        item = " ".join(raw.replace("\n", " ").split())
        item = re.sub(r"^(?:-\s*|\d+\.\s*)", "", item).strip(" ;.")
        if item:
            criteria.append(item)
    return criteria
